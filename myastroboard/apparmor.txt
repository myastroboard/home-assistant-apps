#include <tunables/global>

# The Supervisor rewrites the profile name below to the installed slug
# (e.g. ff6fddf1_myastroboard), so rules must use @{profile_name}, never a literal name.
# Violations show as apparmor="DENIED" in the host audit log:
#   ha host logs --identifier audit --lines 50000 | grep -i apparmor
profile myastroboard flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # entrypoint.sh: chown -R on /data, then setpriv to appuser
  capability chown,
  capability fowner,
  capability dac_override,
  capability dac_read_search,
  capability setuid,
  capability setgid,
  capability kill,

  # gunicorn listener (IPv4 + IPv6) and outbound HTTP/MQTT/DNS
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network netlink raw,

  # Process management between gunicorn master, workers and threads; stop from Supervisor
  signal (send, receive) peer=@{profile_name},
  signal (receive) peer=unconfined,

  # File access: allowed broadly, sensitive areas denied below
  file,

  # Hardening
  deny mount,
  deny umount,
  deny pivot_root,
  deny ptrace,
  deny capability sys_admin,
  deny capability sys_module,
  deny capability sys_rawio,
  deny capability net_admin,
  deny capability sys_ptrace,
  # inet/inet6 only: a bare "deny network raw" would also cancel "network netlink raw" above
  deny network inet raw,
  deny network inet6 raw,
  deny network packet,
  deny /proc/sys/kernel/** w,
  deny /sys/** w,
}
